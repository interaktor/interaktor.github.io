三·删除列表并更新相应的数值

我们现在的功能已经可以添加数值并实时更新了，但是基础的删除功能只能删除，数值并没有因为列表的消失而减少，所以我们要删除列表时候并且删除对应这条列表它里面的值

//删除事件
 ul.addEventListener('click', function (e) {
        //判断当前点击的是不是删除按钮
        if (e.target.classList.contains('history-item_del')) {
          //这里和todolist有相似的地方，我们要找到当前点击的li并且找到它里面的值
          //1.获取当前点击的li
          var currentList = e.target.parentNode.parentNode;
          //2.以这个当前点击的li为基础，去找到它里面的Amount和Type的值！
          //也就是当我们点击一个删除按钮时，它会获取最外层的li这个父元素，然后以父元素为基础去里面找到当前点击的li里面的Type和Amount，把它们的值拿出来，然后进行对应的减少
          var currentAmount = currentList.querSelector('.Amount').innerHTML;
        	//这里的.innerHTML就是获取当前页面里面的值，这里是直接获取的意思，给前面这个变量。如果加上 = 就是赋值的意思
          var currentType = currentList.querySelector('.Type').innerHTML;
          //3.得到类型和值后，要进行和添加【一样】的判断
          if(currentType === '[Income]'){ //这里是值，不是value，值里面是什么样就填什么
            //如果删除的这个列表它的类型是Income那么就让全局变量income 减去我当前点击的这个列表里面的Amount
            income -= parseFloat(currentAmount);
          }  else if(currentType === '[Expenses]') {
            //删除的类型是Expense就让全局变量expense 减去点击的这个list(列表)里面的Amount
            expenses -= parseFloat(currentAmount);
          }
          
          //5.最后删除
          ul.removeChildren(e.target.parentNode.parentNode);
          //6.不要忘了实时更新三大数值
          upateSummary();
        }
    })
