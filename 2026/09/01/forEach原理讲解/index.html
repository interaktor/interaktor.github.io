forEach执行原理讲解

Posted at 2026-09-01
#JavaScript #Study

今天主要讲一下forEach以及里面的新方法，因为这个东西让我苦恼了二次，所以今天打算把这个讲清楚，主要拿「电梯案例」来讲一下forEach

   // 1. 获取所有的内容模块和导航模块
    const sections = document.querySelectorAll('.section');
    const navItems = document.querySelectorAll('.elevator-item');
  
    // 功能一：点击导航按钮，页面平滑滚动到对应板块
//遍历并绑定所有的navItems
    navItems.forEach((item, index) => {
        item.addEventListener('click', () => {
          	//用变量把当前的「内容模块」获取过来
          	//因为「内容模块」和「导航模块」数量都是相同的，所以当我们当前了第二个navItem，它就会获取当前navItem的位置（数字），也就是1，然后sections[1]也就对应的内容模块的第1个，因为它们数量和位置都是对齐的，所以当前navItem的位置就是sections的位置
            const targetSection = sections[index];
          
            // 新方法：scrollIntoView()意思就是「滚动直到显示页面」
            targetSection.scrollIntoView({ behavior: 'smooth' });
        });
    });

 
    // 功能二：页面滚动时，自动计算当前在哪个板块，并高亮对应按钮
    window.addEventListener('scroll', () => {
        //获取页面被滚动的高度（兼容写法）
        const currentScrollTop = document.documentElement.scrollTop || window.pageYOffset;
        //遍历所有的「内容模块」
        sections.forEach((section, index) => {
            //1.获取当前 section 距离顶部的距离
          	//2.这里window的的监听事件是scroll，所以只要有一点点滚动都会触发这个循环（一次循环会把数组的每个值都遍历一次）
          	//3.也就是一次循环就会把这三个sections都会使用一遍，只要有滚动就会触发，每次都是从第0个sections开始，滚动一次就会循环数组中所有的值
            const sectionTop = section.offsetTop; //注意是section,是forEach里面第一个参数
          
          
            // 如果【卷去的高度】达到了【当前板块的高度】（为了提升体验，减去 200px 作为缓冲触发线）
          	//！！！！！！重点！！！！！！！
          	//每次循环每个sections都会判断一次，第一个肯定是满足的，逻辑是这这样的，当第二个条件也满足时，第一个仍然满足，第一个会先启用，但是每次循环都会把这三个section都调试一次，就会出现第一个满足但是在一瞬间，第二个也满足，优先级原理第二个会启用，第一个就没有了，因为数组循环是从0，1，2依次开始的，如果第三个也满足了的话，前面两个还是满足的状态，0和1会分别在一瞬间都会被启用一次(太快肉眼看不见的)，到第三个后，前两个就不显示启用了，因为下方判断里用到了排他思想，每次有满足的都会清除所有，然后只显示最后一个满足条件的，也就是说当currentScollTop超过了所有的sectionTop那么三个条件都满足，但是排他思想只允许有一个显示，只能是所有满足数组值中最后一个满足的才会启用（前面的数值还是满足状态）
            if (currentScrollTop >= sectionTop - 200) {
                // 先移除所有按钮的高亮类名
                navItems.forEach(nav => nav.classList.remove('active'));
                // 给当前索引对应的按钮加上高亮
                navItems[index].classList.add('active');
            }
        });
    });

关于scrollIntoView的一些知识点

//里面主要有两个属性（也可以不填{...}，只不过是默认没有动画跳转）
scrollIntoView({
  behavior:'',
  block:''
})
//behavior属性
behavior:'smooth' 平滑的，带动画效果的移动（最常用）
behavior:'auto' 瞬间移动
behavior:'instant' 瞬间移动
//block属性
block: 'start'：元素的顶部对齐屏幕顶部（最常用）
block: 'center'：让元素停在屏幕的正中间
block: 'end'：元素的底部对齐屏幕底部
