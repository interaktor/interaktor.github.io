Tab栏切换的疑惑点

Posted at 2026-07-16
#JavaScript #Study

今天我来说一下我之前在学习JS上遇到的一个问题

1.首先是获取元素，分成了两大块，一块是tab栏，另一块是tab栏的内容

var lis = document.querySelector('.list').querySelectorAll('li');
var items = document.querySelectorAll('.item')

2.开始循环绑定。我们先给所有的li设置一个索引号(自定义属性)

在给所有li绑定onclick，接下来我们就开始操作
我们一步一步来，当点击任意一个li后，我们先要用到排他思想，就是先「清除所有，再设定自己」用for循环就可以，然后tab栏部分就完成了

for (var i = 0; i < lis.length; i++) {
           //开始的时候就给所有li设定索引号（自定义属性）
           lis[i].setAttribute('index', i)
           lis[i].onclick = function () {
               //1.上面的list我们要用到排他思想，点击一个时就套用current这个类，其他全取消
               //这里我们先把所有li的给清除掉，然后单独再给点击的那一个li设定
               for (var i = 0; i < lis.length; i++) {
                   lis[i].className = ''
               }
               this.className = 'current'
           }
}

3.最后来到内容模块，我们要实现的操作是当上面的tab栏点击哪一个，下面就要显示对应的内容板块

//2.下面内容模块（这里我们点击哪一个就获取哪个的index，然后显示对应的div）
               var index = this.getAttribute('index');
               //这里也要用排他思想先全部清空，然后单独设定内容
               for (var j = 0; j < items.length; j++) {
                   items[j].style.display = 'none'
               }
               items[index].style.display = 'block';

这里要注意，tab栏模块和内容模块要相匹配，它们之间的顺序要一一对应
然后获得我们当前点击的这个li的自定义属性的值
再用到排他思想，先清空所有内容
最后一步很关键，我们要让items的index，也就是它返回来的这个索引号，让这个索引号的items让它显示出来

完整代码如下：

<body>
    <div class="table">
        <div class="list">
            <ul>
                <li class="current">商品介绍</li>
                <li>规格与包装</li>
                <li>售后保障</li>
                <li>商品评价（999+）</li>
                <li>手机社区</li>
            </ul>
        </div>
        <div class="content">
            <div class="item" style="display: block;">
                store
            </div>
            <div class="item">
                packet
            </div>
            <div class="item">
                protect
            </div>
            <div class="item">
                comment
            </div>
            <div class="item">
                unity
            </div>
        </div>
    </div>
    <script>

        //获取元素
        var lis = document.querySelector('.list').querySelectorAll('li');
        var items = document.querySelectorAll('.item')
        //for循环绑定事件
        for (var i = 0; i < lis.length; i++) {
            //开始的时候就给所有li设定索引号（自定义属性）
            lis[i].setAttribute('index', i)
            lis[i].onclick = function () {
                //1.上面的list我们要用到排他思想，点击一个时就套用current这个类，其他全取消
                //这里我们先把所有li的给清除掉，然后单独再给点击的那一个li设定
                for (var i = 0; i < lis.length; i++) {
                    lis[i].className = ''
                }
                this.className = 'current'

                //2.下面内容模块（这里我们点击哪一个就获取哪个的index，然后显示对应的div）
                var index = this.getAttribute('index');
                //这里也要用排他思想先全部清空，然后单独设定内容
                for (var j = 0; j < items.length; j++) {
                    items[j].style.display = 'none'
                }
                items[index].style.display = 'block';

            }

        }
    </script>
</body>
